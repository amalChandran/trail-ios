// Shared demo geometry and imports: see the compiled IntegrationExamples source.

let deliveryTrail = TrailEffect {
    Stroke(.blue, width: 6)
    Reveal(duration: .seconds(2))
}

struct ExistingSwiftUIMapExample: View {
    @State private var playback = TrailPlayback(effect: deliveryTrail)
    private let geometry = TrailMapGeometry(ExampleGeometry.route)
    private let region = MKCoordinateRegion(
        center: CLLocationCoordinate2D(latitude: 37.783, longitude: -122.417),
        span: MKCoordinateSpan(latitudeDelta: 0.015, longitudeDelta: 0.015)
    )
    var body: some View {
        Map(initialPosition: .region(region)) {
            TrailMapContent(geometry: geometry, playback: playback)
        }
        .trailPlayback(playback)
    }
}
